    }
    meta.lastOffers = offers.map((relic) => relic.id);
    return offers;
  }

    const meta = ensureMeta();

        </article>
      `;
    }).join("");
    modalBody.innerHTML = `
      <div class="build-summary"><b>SCRAP AVAILABLE</b><b>${meta.scrap}</b></div>
      <div class="rogue-cards">${cards}</div>
      <div class="rogue-footer-actions">
        <button class="rogue-wide-button secondary" id="relicReroll" type="button" ${meta.scrap >= 3 ? "" : "disabled"}>REROLL · 3 SCRAP</button>

      button.addEventListener("click", () => buyRelic(button.dataset.relicBuy));
    }

    });
    modalBody.querySelector("#relicSkip").addEventListener("click", closeModal);
  }

  function buyRelic(id) {
    const relic = relicById(id);

    meta.relicDiscount = 0;
    applyImmediateRelic(id);
    toast(`${relic.name} acquired`);
    closeModal();
    render();
  }

  function heroOf(type) {
    return game.heroes.find((hero) => hero.alive && hero.type === type) || null;
  }

  function adjacent(first, second) {
    return Boolean(first && second && manhattan(first, second) === 1);
  }

  function aligned(first, second) {
    return Boolean(first && second && (first.r === second.r || first.c === second.c));
  }

  function contiguousLine(heroes) {
    for (let row = 0; row < SIZE; row += 1) {
      const columns = heroes.filter((hero) => hero.r === row).map((hero) => hero.c).sort((a, b) => a - b);
      if (columns.length >= 4 && columns[columns.length - 1] - columns[0] <= columns.length - 1) return true;
    }
    for (let column = 0; column < SIZE; column += 1) {
      const rows = heroes.filter((hero) => hero.c === column).map((hero) => hero.r).sort((a, b) => a - b);
      if (rows.length >= 4 && rows[rows.length - 1] - rows[0] <= rows.length - 1) return true;
    }
    return false;
  }

  function detectTactics() {
    if (!game) return { formation: null, combos: [] };
    const heroes = game.heroes.filter((hero) => hero.alive);
    const blade = heroOf("blade");
    const shield = heroOf("shield");
    const spear = heroOf("spear");
    const bow = heroOf("bow");
    const banner = heroOf("banner");
    const crossKeys = new Set([
      key(BEACON.r, BEACON.c),
      key(BEACON.r - 1, BEACON.c),
      key(BEACON.r + 1, BEACON.c),
      key(BEACON.r, BEACON.c - 1),
      key(BEACON.r, BEACON.c + 1)
    ]);
    const aroundKeys = new Set([...crossKeys].filter((value) => value !== key(BEACON.r, BEACON.c)));
    const heroKeys = new Set(heroes.map((hero) => key(hero.r, hero.c)));
    let formation = null;
    if (heroes.length === 5 && [...crossKeys].every((value) => heroKeys.has(value))) {
      formation = FORMATION_RULES[0];
    } else if (heroes.length >= 4 && [...aroundKeys].every((value) => heroKeys.has(value))) {
      formation = FORMATION_RULES[1];
    } else if (contiguousLine(heroes)) {
      formation = FORMATION_RULES[2];
    } else if (shield && heroes.filter((hero) => hero.id !== shield.id && adjacent(hero, shield)).length >= 2) {
      formation = FORMATION_RULES[3];
    } else if (blade && shield && spear && manhattan(blade, shield) <= 2 && manhattan(spear, shield) <= 2 && manhattan(blade, spear) <= 3) {
      formation = FORMATION_RULES[4];
    }
    const combos = [];
    if (adjacent(blade, shield)) combos.push(COMBO_RULES[0]);
    if (aligned(bow, spear) && manhattan(bow, spear) <= 4) combos.push(COMBO_RULES[1]);
    if (adjacent(banner, shield)) combos.push(COMBO_RULES[2]);
    if (blade && bow && manhattan(blade, bow) <= 2) combos.push(COMBO_RULES[3]);
    if (adjacent(spear, shield)) combos.push(COMBO_RULES[4]);
    return { formation, combos };
  }

        if (hero) hero.guard = Math.max(hero.guard, formationPower);
      }

      const shield = heroOf("shield");
