  function actNumber(round = game?.round || 1) {
    return Math.max(1, Math.floor((round - 1) / ACT_LENGTH) + 1);
  }

  function actInfo(round = game?.round || 1) {
    const number = actNumber(round);
    if (number <= ACTS.length) {
      return { ...ACTS[number - 1], number, endlessTier: 0 };
    }
    const endlessIndex = (number - ACTS.length - 1) % ENDLESS_ACTS.length;
    const endlessTier = number - ACTS.length;
    const source = ENDLESS_ACTS[endlessIndex];
    return {
      ...source,
      number,
      endlessTier,
      name: `${source.name} ${toRoman(Math.ceil(endlessTier / ENDLESS_ACTS.length))}`
    };
  }

  function toRoman(value) {
    const table = [
      [10, "X"], [9, "IX"], [5, "V"], [4, "IV"], [1, "I"]
    ];
    let remaining = Math.max(1, value);
    let result = "";
    for (const [amount, glyph] of table) {
      while (remaining >= amount) {
        result += glyph;
        remaining -= amount;
      }
    }
    return result;
  }
