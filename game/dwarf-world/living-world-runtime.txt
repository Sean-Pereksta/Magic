function lmCamp(s){
  const known=lm.factions.find(v=>v.id===s.id);
  if(known)return known;
  if(lm.factions.length>=256||!tSeen[idx(s.x,s.y)])return null;
  const depth=depthAt(s.x,s.y);
  const names=depth>620?['abyss','undead','orc']:depth>260?['orc','undead','brood']:['goblin','brood'];
  const type=names[Math.abs(s.x*13+s.y*7)%names.length];
  const camp={id:s.id,type,x:s.x,y:s.y,phase:'gathering',population:0,stock:0,nextRaid:lm.time+18,raids:0,discovered:true,knowledge:null};
  lm.factions.push(camp);
  lmChronicle(LM_FACTIONS[type].name+' uncovered at depth '+depth+'.');
  return camp;
}

  if(lm.settlements.length<12)lmSeedSettlement();

  }

}
// Enemies know only what they or their scouts have observed along clear sight.
nearestEnemyTarget=function(e){
  const range=e.lmRole==='scout'?18:13,r2=range*range;
  const candidates=[...lmNearbyDwarves(e.x,e.y,range+1).filter(d=>d.hp>0).map(ref=>({kind:'dwarf',x:Math.round(ref.x),y:Math.round(ref.y),ref})),...towers.map(ref=>({kind:'tower',x:ref.x,y:ref.y,ref})),...depots.map(ref=>({kind:'depot',x:ref.x,y:ref.y,ref})),{kind:'fortress',x:fortress.x,y:fortress.y,ref:fortress}];
  let best=null,bd=Infinity;for(const t of candidates){const q=dist2(e.x,e.y,t.x,t.y);if(q>r2||!lmSight(e.x,e.y,t.x,t.y))continue;const score=q*(e.lmRole==='sapper'&&t.kind!=='dwarf'?.5:1);if(score<bd){bd=score;best=t}}
  if(best){
    e.lmMemory={x:best.x,y:best.y,until:lm.time+25};
    if(e.lmRole==='scout'&&e.lmHome&&!e.lmScoutReport&&(e.lmScoutAgainAt||0)<=lm.time){
      e.lmScoutReport={x:best.x,y:best.y,readyAt:lm.time+1.2};
      e.lmRetreatUntil=lm.time+8;
      return{kind:'retreat',x:e.lmHome.x,y:e.lmHome.y,ref:null};
    }
    return best;
  }
  if(e.lmMemory?.until>lm.time)return{kind:'memory',x:e.lmMemory.x,y:e.lmMemory.y,ref:null};
  if(e.lmHome){const dx=((Math.floor(lm.time/8)+String(e.id).length)%3)-1,dy=dx===0?1:0;const x=e.lmHome.x+dx,y=e.lmHome.y+dy;if(inBounds(x,y)&&isWalkable(tType[idx(x,y)]))return{kind:'patrol',x,y,ref:null}}
  return null;
};
const LM_SAPPER_DIRECTIONS=[[0,0],[0,-1],[1,0],[0,1],[-1,0]];
const LM_BREACHABLE=new Set(['gate','wall','barricade','slit']);
function lmSapperTick(enemy){
  const x=Math.round(enemy.x),y=Math.round(enemy.y);
  const breach=enemy.lmBreach;
  if(breach){
    const defense=lmEngine.infraAt(breach.x,breach.y);
    if(!defense||defense.id!==breach.id||Math.abs(x-breach.x)+Math.abs(y-breach.y)>1){enemy.lmBreach=null;return false}
    if(lm.time<breach.until)return true;
    enemy.lmBreach=null;
    enemy.lmBreachAt=lm.time+1;
    defense.hp-=enemy.dmg*4;
    fxText(defense.x,defense.y,'BREACH','#ffb06c');
    if(defense.hp<=0){
      lm.infrastructure=lm.infrastructure.filter(item=>item!==defense);
      lmEngine.reindex();
      window.dispatchEvent(new CustomEvent('dwarfworld:terrain',{detail:{x:defense.x,y:defense.y}}));
      lmChronicle('A sapper breached a '+defense.type+'.');
    }
    return true;
  }
  if((enemy.lmBreachAt||0)>lm.time)return false;
  for(const [dx,dy] of LM_SAPPER_DIRECTIONS){
    if(!inBounds(x+dx,y+dy))continue;
    const defense=lmEngine.infraAt(x+dx,y+dy);
    if(!defense||!LM_BREACHABLE.has(defense.type))continue;
    enemy.path=null;
    enemy.lmBreach={id:defense.id,x:defense.x,y:defense.y,until:lm.time+1.8};
    return true;
  }
  return false;
}
const lmBaseEnemy=updateEnemy;
updateEnemy=function(e,dt){if(!e||e.hp<=0)return;
  const infra=lmEngine.infraAt(Math.round(e.x),Math.round(e.y));if(infra?.type==='trap'&&(infra.readyAt||0)<lm.time){infra.readyAt=lm.time+12;applyEnemyDamage(e,Math.max(10,e.maxHp*.15));e.stunLeft=Math.max(e.stunLeft||0,1.2)}
  if(e.hp<=0)return;
  if(e.stunLeft>0){
    if(e.lmBreach)e.lmBreachAt=lm.time+1;
    if(e.lmTell)e.lmAbilityAt=lm.time+2;
    if(e.lmShot)e.lmShotAt=lm.time+2;
    e.lmBreach=null;
    e.lmTell=null;
    e.lmShot=null;
    lmBaseEnemy(e,dt);
    return;
  }
  if(e.lmRole==='sapper'&&lmSapperTick(e))return;
  e.lmTacticalClock=(e.lmTacticalClock||0)-dt;
  if(e.lmTacticalClock<=0){e.lmTacticalClock=1.5;if(e.lmHome&&!e.lmBoss&&e.hp<e.maxHp*.25&&!(e.lmRetreatUntil>lm.time)){e.lmRetreatUntil=lm.time+5;e.path=null}
    if(e.lmRole==='commander')for(const ally of lmNearbyEnemies(e.x,e.y,6))if(ally!==e&&dist2(e.x,e.y,ally.x,ally.y)<36&&lmSight(e.x,e.y,ally.x,ally.y))ally.lmCommandUntil=lm.time+2;
  }
  if(e.lmRetreatUntil>lm.time&&e.lmHome){
    const home=e.lmHome;
    if(dist2(e.x,e.y,home.x,home.y)<1){
      e.path=null;
      e.hp=Math.min(e.maxHp,e.hp+dt*2);
      if(e.lmScoutReport?.readyAt<=lm.time){
        const camp=lm.factions.find(c=>c.id===e.lmCamp);
        if(camp){const {x,y}=e.lmScoutReport;camp.knowledge={x,y,until:lm.time+60}}
        e.lmScoutReport=null;
        e.lmScoutAgainAt=lm.time+12;
        e.lmRetreatUntil=0;
      }
    }else{
      if(!e.path?.length&&(e.lmReturnPathAt||0)<=lm.time){
        e.lmReturnPathAt=lm.time+.6;
        e.path=bfsToPoint(Math.round(e.x),Math.round(e.y),home.x,home.y,0,true);
      }
      stepAlongPath(e,dt);
    }
    return;
  }
  e.lmScoutReport=null;
  if(e.lmBoss&&lmBossTick(e))return;
  // Ranged support takes a clear firing position and spends time charging;
  // the sight test is repeated at impact so a newly shut gate protects dwarves.
  if(!e.lmBoss&&!(e.phasedLeft>0)&&(e.type==='shaman'||e.type==='phaser')){
    e.lmShotAt=e.lmShotAt||lm.time+3;
    if(e.lmShot){if(lm.time<e.lmShot.until)return;const d=dwarves.find(d=>d.id===e.lmShot.id);if(d&&d.hp>0&&dist2(e.x,e.y,d.x,d.y)<64&&lmSight(e.x,e.y,d.x,d.y)){d.hp-=Math.max(1,Math.round(e.dmg*(1-(d.defense||0))));fxBeam(e.x,e.y,d.x,d.y,'rgba(166,190,255,.9)');markUI('selected','dwarves','hud')}e.lmShot=null;e.lmShotAt=lm.time+4}
    else if(lm.time>=e.lmShotAt){const d=lmNearbyDwarves(e.x,e.y,8).find(d=>d.hp>0&&dist2(e.x,e.y,d.x,d.y)>5&&dist2(e.x,e.y,d.x,d.y)<64&&lmSight(e.x,e.y,d.x,d.y));if(d){e.lmShot={id:d.id,until:lm.time+1.2};e.path=null;return}e.lmShotAt=lm.time+1}
  }
  const dmg=e.dmg;if(e.lmCommandUntil>lm.time)e.dmg=Math.round(dmg*1.15);try{lmBaseEnemy(e,dt)}finally{e.dmg=dmg}
};

const lmBaseSpecial=tickEnemySpecials;
tickEnemySpecials=function(e,dt){
  const aura=e.auraRegen,cap=e.summonCap,start=enemies.length;
  e.auraRegen=0;
  if(e.type==='necromancer'&&(e.summonCd<=dt||e.lmBoss)){
    e.summonCap=e.lmBoss?0:Math.min(cap||0,Math.max(0,240-start+countMinionsOwnedBy(e.id)));
  }
  try{lmBaseSpecial(e,dt)}finally{e.auraRegen=aura;e.summonCap=cap}
  for(let n=start;n<enemies.length;n++){
    enemies[n].lmCamp=e.lmCamp;
    enemies[n].lmHome=e.lmHome;
  }
  if(!(aura>0))return;
  for(const ally of lmNearbyEnemies(e.x,e.y,e.auraRange)){
    if(ally!==e&&ally.hp>0&&ally.hp<ally.maxHp&&dist2(e.x,e.y,ally.x,ally.y)<e.auraRange**2&&lmSight(e.x,e.y,ally.x,ally.y)){
      ally.hp=Math.min(ally.maxHp,ally.hp+aura*dt);
    }
  }
};
function lmTellHits(tell,attack,unit){
  const dx=unit.x-tell.x,dy=unit.y-tell.y;
  if(attack.shape!=='lane')return dx*dx+dy*dy<=attack.radius**2;
  const forward=dx*tell.dx+dy*tell.dy;
  const side=dx*tell.dy-dy*tell.dx;
  return forward>=0&&forward<=attack.length&&Math.abs(side)<=attack.halfWidth;
}
function lmBossTick(enemy){
  const attack=LM_BOSS_ATTACKS[enemy.lmBoss];
  if(!attack)return false;
  const tell=enemy.lmTell;
  if(tell){
    if(lm.time<tell.until)return true;
    enemy.lmTell=null;
    enemy.lmAbilityAt=lm.time+11;
    if(attack.minion){
      let slots=Math.min(2,240-enemies.length,6-countMinionsOwnedBy(enemy.id));
      while(slots-->0){
        const child=makeEnemy(Math.round(enemy.x),Math.round(enemy.y),enemy.sourceDepth);
        lmSetEnemyType(child,attack.minion);
        child.summonerId=enemy.id;
        child.isMinion=true;
        child.noLoot=true;
        child.lmHome=enemy.lmHome;
        child.lmCamp=enemy.lmCamp;
        enemies.push(child);
      }
    }else{
      for(const dwarf of lmNearbyDwarves(tell.x,tell.y,(attack.length||attack.radius)+1)){
        if(dwarf.hp<=0||!lmTellHits(tell,attack,dwarf)||!lmSight(enemy.x,enemy.y,dwarf.x,dwarf.y))continue;
        const damage=Math.max(1,Math.round(enemy.dmg*1.4*(1-(dwarf.defense||0))));
        dwarf.hp-=damage;
        fxText(dwarf.x,dwarf.y,`-${damage}`,'#ffb06c');
        markUI('selected','dwarves','hud');
      }
    }
    return true;
  }
  if(lm.time<(enemy.lmAbilityAt||0))return false;
  const target=nearestEnemyTarget(enemy);
  if(!target?.ref){enemy.lmAbilityAt=lm.time+1;return false}
  const onTarget=attack.shape==='circle';
  const x=onTarget?target.x:enemy.x,y=onTarget?target.y:enemy.y;
  const dx=target.x-enemy.x,dy=target.y-enemy.y,distance=Math.hypot(dx,dy);
  enemy.lmTell={x,y,dx:distance?dx/distance:1,dy:distance?dy/distance:0,until:lm.time+2.5};
  enemy.path=null;
  return true;
}

function lmDrawBossTell(tell,attack){
  const point=worldToScreen(tell.x,tell.y);
  const s=point.s,x=point.x+s*.5,y=point.y+s*.5;
  const summon=attack.shape==='summon';
  ctx.fillStyle=summon?'rgba(175,132,255,.24)':'rgba(238,94,48,.25)';
  ctx.strokeStyle=summon?'#dcc4ff':'#ffbe8a';
  ctx.lineWidth=2;
  ctx.beginPath();
  if(attack.shape==='lane'){
    const endX=x+tell.dx*attack.length*s,endY=y+tell.dy*attack.length*s;
    const sideX=-tell.dy*attack.halfWidth*s,sideY=tell.dx*attack.halfWidth*s;
    ctx.moveTo(x+sideX,y+sideY);
    ctx.lineTo(endX+sideX,endY+sideY);
    ctx.lineTo(endX-sideX,endY-sideY);
    ctx.lineTo(x-sideX,y-sideY);
    ctx.closePath();
  }else ctx.arc(x,y,attack.radius*s,0,Math.PI*2);
  ctx.fill();
  ctx.stroke();
  if(attack.shape==='stomp'||summon){
    ctx.beginPath();
    ctx.arc(x,y,attack.radius*s*.72,0,Math.PI*2);
    ctx.setLineDash(summon?[2,4]:[6,3]);
    ctx.stroke();
    ctx.setLineDash([]);
  }
  const remaining=Math.max(0,Math.min(1,(tell.until-lm.time)/2.5));
  ctx.fillStyle='#16141c';
  ctx.fillRect(x-16,y-5,32,5);
  ctx.fillStyle=summon?'#dcc4ff':'#ffbe8a';
  ctx.fillRect(x-15,y-4,30*remaining,3);
}
function lmDrawEnemyWarnings(){
  let drawn=0;
  for(const enemy of enemies){
    if(enemy.hp<=0||!(enemy.lmBoss||enemy.lmBreach||enemy.lmShot||enemy.lmScoutReport))continue;
    if(!enemy.lmBoss&&drawn>=14)continue;
    const x=Math.round(enemy.x),y=Math.round(enemy.y);
    if(!inBounds(x,y)||tVis[idx(x,y)]!==visMark)continue;
    const rendered=typeof livingRenderPosition==='function'?livingRenderPosition(enemy):enemy;
    const p=worldToScreen(rendered.x,rendered.y);
    if(p.x<-p.s*2||p.y<-p.s*2||p.x>innerWidth||p.y>innerHeight)continue;
    drawn++;
    const attack=LM_BOSS_ATTACKS[enemy.lmBoss];
    if(attack){
      if(enemy.lmTell)lmDrawBossTell(enemy.lmTell,attack);
      lmDrawCombatLabel(enemy.lmTell?attack.name:enemy.lmBoss,p.x+p.s*.5,p.y-8,'#ffd5aa');
    }
    if(enemy.lmBreach){
      const q=worldToScreen(enemy.lmBreach.x,enemy.lmBreach.y);
      ctx.strokeStyle='#ffbe8a';
      ctx.lineWidth=2;
      ctx.setLineDash([4,3]);
      ctx.strokeRect(q.x+1,q.y+1,q.s-2,q.s-2);
      ctx.setLineDash([]);
      lmDrawCombatLabel('Breach',q.x+q.s*.5,q.y-4,'#ffd5aa');
    }
    if(enemy.lmShot){
      const target=dwarves.find(d=>d.id===enemy.lmShot.id&&d.hp>0);
      if(target&&lmSight(enemy.x,enemy.y,target.x,target.y)){
        const q=worldToScreen(target.x,target.y);
        ctx.strokeStyle='#d8d0ff';
        ctx.lineWidth=2;
        ctx.setLineDash([4,4]);
        ctx.beginPath();
        ctx.moveTo(p.x+p.s*.5,p.y+p.s*.5);
        ctx.lineTo(q.x+q.s*.5,q.y+q.s*.5);
        ctx.stroke();
        ctx.setLineDash([]);
      }
      lmDrawCombatLabel('Charging',p.x+p.s*.5,p.y-4,'#d8d0ff');
    }
    if(enemy.lmScoutReport)lmDrawCombatLabel('Reporting',p.x+p.s*.5,p.y-4,'#ffe3a8');
  }
}
