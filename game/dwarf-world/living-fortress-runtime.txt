function lmSight(ax,ay,bx,by){
  let x=Math.round(ax),y=Math.round(ay),tx=Math.round(bx),ty=Math.round(by);
  const dx=Math.abs(tx-x),dy=-Math.abs(ty-y),sx=x<tx?1:-1,sy=y<ty?1:-1;let err=dx+dy;
  for(let n=0;n<96;n++){
    if(x===tx&&y===ty)return true;
    const e=err*2;if(e>=dy){err+=dy;x+=sx}if(e<=dx){err+=dx;y+=sy}
    if(!inBounds(x,y)||!isWalkable(tType[idx(x,y)]))return false;
    // Arrow slits allow sight, solid walls and shut gates do not.
    const v=lmEngine.infraAt(x,y);if(v&&(v.type==='wall'||v.type==='gate'&&lm.readiness==='lockdown'))return false;
  }return false;
}

const lmBaseDead=removeDead;
removeDead=function(){
  // Keep a dead carrier's real position/cargo in the save if the pile cap is
  // full. It is inert, consumes its existing population slot, and cannot be
  // repeatedly resurrected by the legacy death wrapper while awaiting space.
  const held=[],dying=[];
  for(let i=dwarves.length-1;i>=0;i--){const d=dwarves[i];if(d._livingRemains){dwarves.splice(i,1);const dropped=lmEngine.dropCargo(d),canceled=lmEngine.cancel(d);if(!dropped||!canceled)held.push(d)}else if(d.hp<=0){lmEngine.dropCargo(d);lmEngine.cancel(d);dying.push(d)}}
  lmBaseDead();
  for(const d of dying)if(d.hp<=0&&(d.living?.cargo||d.living?.job?.started)&&!dwarves.includes(d)){d._livingRemains=true;d.living.activity='Cargo waiting for storage space';held.push(d)}
  dwarves.push(...held);
};

saveGameAsync=function(){
  if(lmLoading){toast('Save failed: the world is still loading.');return Promise.resolve(false)}
  if(lmSaving)return lmSaving;
  lmSaving=(async()=>{try{
    const cleanDwarf=d=>{const v=pruneEntity(d);if(v.living){v.living={...v.living};delete v.living.path;delete v.living.retry;delete v.living.nextThink}return v};
    const save={v:2,W,H,fortress:{...fortress},cam:{...cam},gold,recruitCount,dwarfCounter,forge:forge?{...forge}:null,depots:depots.map(pruneEntity),towers:towers.map(pruneEntity),spawners:spawners.map(pruneEntity),enemies:enemies.map(pruneEntity),groundLoot:groundLoot.map(L=>({id:L.id,x:L.x,y:L.y,payload:L.payload})),loot:loot.map(pruneEntity),dwarves:dwarves.map(cleanDwarf),tTypeB64:_u8ToB64(new Uint8Array(tType.buffer)),tHpB64:_u8ToB64(new Uint8Array(tHp.buffer)),tSeenB64:_u8ToB64(new Uint8Array(tSeen.buffer)),fortressExpansion:JSON.parse(JSON.stringify(dwx)),livingMountain:lmEngine.serialize()};
    const packed=SAVE_MAGIC+await gzipStrToB64(JSON.stringify(save));localStorage.setItem(SAVE_KEY,packed);toast('Saved locally. (~'+Math.round(packed.length/1024)+' KB)');return true;
  }catch(error){console.error('[Living Mountain save]',error);toast('Save failed; previous local save was preserved.');return false}finally{lmSaving=null}})();return lmSaving;
};

// Validate nested state before the legacy loader commits its tile/entity arrays.
// Routes are transient; persisted cargo and unfinished production remain intact.
function lmValidateRestore(save,mountain){
  const record=v=>!!v&&typeof v==='object'&&!Array.isArray(v),ref=v=>typeof v==='string'&&v.length>0||Number.isFinite(v);
  const fail=part=>{throw new Error('Save corrupted ('+part+').')};
  const point=v=>record(v)&&Number.isFinite(v.x)&&Number.isFinite(v.y)&&inBounds(Math.round(v.x),Math.round(v.y));
  const material=v=>window.DwarfLivingSimulation.MATERIALS.includes(v);
  const quantity=n=>Number.isFinite(n)&&n>0&&n<=1000000;
  const item=(v,part)=>{if(!record(v)||!['armor','weapon','pickaxe'].includes(v.category)||typeof v.name!=='string'||v.rarity!==undefined&&!['string','number'].includes(typeof v.rarity))fail(part)};
  const locations=new Map([...mountain.rooms,...mountain.stores,...mountain.infrastructure,...mountain.deposits,...mountain.settlements].map(v=>[v.id,v]));
  if(save.cam!==undefined&&(!record(save.cam)||!point(save.cam)||!Number.isFinite(save.cam.zoom)||save.cam.zoom<=0))fail('camera');
  for(const v of save.loot||[])item(v,'inventory item');
  for(const d of save.dwarves){
    for(const slot of ['armor','weapon','pickaxe'])if(d[slot]!=null)item(d[slot],'dwarf equipment');
    const p=d.living;if(p==null)continue;if(!record(p))fail('dwarf living state');
    if(p.history!==undefined&&(!Array.isArray(p.history)||p.history.some(v=>typeof v!=='string')))fail('dwarf history');
    if(p.relationships!==undefined&&(!record(p.relationships)||Object.values(p.relationships).some(v=>!Number.isFinite(v))))fail('dwarf relationships');
    if(p.cargo!=null&&(!record(p.cargo)||!material(p.cargo.resource)||!quantity(p.cargo.amount)))fail('dwarf cargo');
    const j=p.job;if(j==null)continue;
    if(!record(j)||!['haul','deliver','dig','construct','produce','eat','sleep','social','recover','equip','patrol','rescue','explore','shelter'].includes(j.kind)||!point(j.goal)||!Number.isInteger(j.goal.x)||!Number.isInteger(j.goal.y))fail('dwarf job');
    if(j.elapsed!==undefined&&(!Number.isFinite(j.elapsed)||j.elapsed<0)||j.claim!==undefined&&typeof j.claim!=='string')fail('dwarf job progress');
    if(j.kind==='rescue'){if(!ref(j.patient))fail('rescue target')}else if(!ref(j.dest))fail('job destination');
    if(j.kind==='haul'&&(!ref(j.source)||!material(j.resource)||!quantity(j.amount)))fail('hauling job');
    // A legitimately removed target is canceled by the simulation; an existing
    // target of the wrong kind must not reach inventory/recipe consumers.
    const destination=locations.get(j.dest),source=locations.get(j.source);
    if(destination&&!['dig','rescue','explore'].includes(j.kind)&&!record(destination.inventory))fail('job destination inventory');
    if(j.kind==='haul'&&source&&!record(source.inventory))fail('job source inventory');
    if(j.kind==='produce'&&destination&&!LM_ROOMS[destination.type]?.recipe)fail('production workplace');
    if(j.kind==='construct'&&destination&&!mountain.rooms.includes(destination))fail('construction room');
    for(const k of ['started','completed'])if(j[k]!==undefined&&typeof j[k]!=='boolean')fail('production progress');
    if(j.inputs!==undefined&&(!record(j.inputs)||Object.entries(j.inputs).some(([k,n])=>!material(k)||!quantity(n))))fail('production ingredients');
  }
  if(!mountain.stores.length)fail('missing fortress stockpile');
  for(const name of ['rooms','infrastructure','deposits','stores','designations','settlements'])for(const v of mountain[name])if(!record(v)||!ref(v.id))fail(name+' identity');
  for(const name of ['deposits','stores'])for(const v of mountain[name])if(!record(v.inventory))fail(name+' inventory');
  for(const v of mountain.chronicle)if(!record(v)||typeof v.text!=='string')fail('chronicle entry');
  for(const v of mountain.factions)if(!point(v)||!ref(v.id)||!LM_FACTIONS[v.type]||v.knowledge!=null&&!point(v.knowledge))fail('faction');
  for(const c of mountain.caravans){
    if(!point(c)||!point(c.goal)||!ref(c.id)||!ref(c.source)||!ref(c.dest)||!material(c.resource)||!quantity(c.amount)||!Number.isFinite(c.hp)||!Number.isFinite(c.speed)||c.speed<=0)fail('caravan');
    c.path=null;c.retry=0;
  }
  const extension=save.fortressExpansion;if(extension!=null){
    if(!record(extension))fail('fortress extension');
    for(const v of extension.surveys||[])if(!point(v)||!Number.isFinite(v.expires)||!Number.isFinite(v.radius)||v.radius<0)fail('survey');
    if(extension.stats!==undefined&&!record(extension.stats))fail('fortress statistics');
    for(const b of extension.buildings||[])if(!point(b)||b.stats!==undefined&&!record(b.stats))fail('civic building statistics');
  }
}
loadGameAsync=async function(){if(lmLoading)return false;if(lmSaving)await lmSaving;lmLoading=true;const announce=toast;let loaded=false;try{

  if(!save||![1,2].includes(save.v??1)||save.W!==W||save.H!==H)throw new Error('Save incompatible with this world size.');
  const migrated=window.DwarfLivingSimulation.migrate(save.livingMountain,fortress,W,H);
  // Validate the tile lengths before permitting any core state mutation.
  for(const[name,expected]of [[save.v===2?'tTypeB64':'tType',tType.byteLength],[save.v===2?'tHpB64':'tHp',tHp.byteLength]])if(_b64ToU8(save[name]).length!==expected)throw new Error('Save corrupted (tile data size mismatch).');
  if(!save.fortress||!Number.isFinite(save.fortress.hp)||!Number.isFinite(save.fortress.maxHp)||!Array.isArray(save.dwarves))throw new Error('Save corrupted (fortress data).');
  for(const name of ['depots','towers','spawners','enemies','groundLoot','loot','dwarves']){

  }

  if(save.fortressExpansion){for(const name of ['buildings','protectedIds','relicsFound','presentedRelics','surveys'])if(save.fortressExpansion[name]!==undefined&&!Array.isArray(save.fortressExpansion[name]))throw new Error('Save corrupted (fortress '+name+').');for(const b of save.fortressExpansion.buildings||[])if(!b||!DWX_BUILDINGS[b.type]||!Number.isFinite(b.x)||!Number.isFinite(b.y))throw new Error('Save corrupted (civic building).')}

  await dwxBaseLoadAsync();

  dwx=Object.assign(dwxFresh(),save.fortressExpansion||{});for(const k of ['buildings','protectedIds','relicsFound','presentedRelics','surveys'])if(!Array.isArray(dwx[k]))dwx[k]=[];
  for(const b of dwx.buildings)if(!b.stats)b.stats={serviced:0,goldReturned:0};
  lm=lmEngine.restore(migrated);lmWorldReset();dwxRefreshBuild();for(const it of loot)if(it.relicId)dwxProtect(it.id,true);window.dispatchEvent(new CustomEvent('dwarfworld:terrain',{detail:{}}));markUI('all');announce('Loaded local save.');return true;
}catch(error){console.error('[Living Mountain load]',error);announce('Load failed: '+error.message);return false}finally{toast=announce;lmLoading=false}};
